Analysis - //NAME OF GRAPH = 
VAR FirstYear = MIN('//NAME OF TABLE'[//NAME OF COL])
VAR LastYear = MAX('//NAME OF TABLE'[//NAME OF COL])

VAR FirstValue = 
    CALCULATE(
        [Average // NAME OF GRAPH],
       '//NAME OF TABLE'[//NAME OF COL] = FirstYear
    )
    
VAR LastValue = 
    CALCULATE(
        [Average //NAME OF GRAPH],
       '//NAME OF TABLE'[//NAME OF COL] = LastYear
    )

VAR PctChange = DIVIDE(LastValue - FirstValue, FirstValue, 0)

RETURN
    SWITCH(
        TRUE(),
        PctChange >= 0.05, "improving",
        PctChange <= -0.05, "declining",
        "stable"
    )
