Analysis - Overdue Period Change % = 
VAR StartDate =
    CALCULATE (
        MIN ( tblDepartmentSnapshots[SnapshotDate] ),
        ALLSELECTED ( tblDepartmentSnapshots[SnapshotDate] )
    )
VAR EndDate =
    CALCULATE (
        MAX ( tblDepartmentSnapshots[SnapshotDate] ),
        ALLSELECTED ( tblDepartmentSnapshots[SnapshotDate] )
    )
VAR StartValue =
    CALCULATE (
        SUM ( tblDepartmentSnapshots[Overdue] ),
        tblDepartmentSnapshots[SnapshotDate] = StartDate
    )
VAR EndValue =
    CALCULATE (
        SUM ( tblDepartmentSnapshots[Overdue] ),
        tblDepartmentSnapshots[SnapshotDate] = EndDate
    )
RETURN
    DIVIDE ( EndValue - StartValue, EndValue )
